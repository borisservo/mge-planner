<script>
  const resNormal = { 1: 54000, 2: 216000, 3: 540000, 4: 912000, 5: 2160000, 6: 3780000, 7: 5292000, 8: 7392000 };
  const resRich = { 2: 720000, 3: 1800000, 4: 3040000, 5: 7200000, 6: 12600000, 7: 17640000, 8: 24640000 };
  const baseTimes = { 1: 18, 2: 72, 3: 180, 4: 288, 5: 648, 6: 1080, 7: 1512, 8: 2016 };
  const richTimes = { 2: 120, 3: 300, 4: 480, 5: 1080, 6: 1800, 7: 2520, 8: 3360 };
  const unitPoints = { 3: 5, 4: 10, 5: 20, 6: 50, 7: 100 };

  let lineupCount = 0;
  let promoCount = 0;

  function switchTab(event, tabId) {
    document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
    document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
    document.getElementById(tabId).classList.add('active');
    event.currentTarget.classList.add('active');
  }
  function addLineup() {
    lineupCount++;
    const container = document.getElementById('lineupContainer');
    const div = document.createElement('div');
    div.className = 'prediction';
    div.id = 'lineup-' + lineupCount;
    div.style.marginBottom = "15px";
    div.innerHTML = `
            <div style="display:flex; justify-content:space-between; align-items:center; border-bottom:1px solid #333; margin-bottom:10px; padding-bottom:5px;">
                <span style="color:var(--gold); font-weight:bold;">Lineup ${lineupCount}</span>
                <div style="display:flex; align-items:center; gap:12px;">
                    <label style="margin:0; display:flex; align-items:center; gap:6px; cursor:pointer; color:var(--blue); font-size:0.8rem;">
                        <input type="checkbox" class="l-rich" onchange="calculate()" style="width:16px; height:16px; margin:0;"> <b>RICH</b>
                    </label>
                    <button onclick="removeLineup(${lineupCount})" style="background:#ff4444; color:white; border:none; border-radius:4px; cursor:pointer; padding:2px 10px;">X</button>
                </div>
            </div>
            <div style="display:grid; grid-template-columns: 1fr 1fr; gap:10px;">
                <div><label>Level</label><select class="l-lvl" onchange="calculate()">${[8,7,6,5,4,3,2,1].map(l => `<option value="${l}">Lvl ${l}</option>`).join('')}</select></div>
                <div><label>Rounds</label><input type="number" class="l-rounds" value="1" oninput="calculate()"></div>
                <div><label>Speed%</label><input type="number" class="l-speed" value="0" oninput="calculate()"></div>
                <div><label>Comp%</label><input type="number" class="l-comp" value="0" oninput="calculate()"></div>
            </div>
            <div style="margin-top:10px; font-size:0.8rem; display:flex; justify-content:space-between; background:#111; padding:8px; border-radius:4px;">
                <span>Total Time: <b class="l-time">--</b></span><span>MGE Points: <b class="l-pts" style="color:var(--gold);">0</b></span>
            </div>`;
    container.appendChild(div);
    calculate();
  }

  function removeLineup(id) { document.getElementById('lineup-' + id).remove(); calculate(); }

  function addPromotion() {
    promoCount++;
    const container = document.getElementById('promoContainer');
    const div = document.createElement('div');
    div.className = 'prediction';
    div.id = 'promo-' + promoCount;
    div.innerHTML = `<div style="display:grid; grid-template-columns: 1fr 1fr 1fr auto; gap:10px; align-items:end;"><div><label>From</label><select class="p-from" onchange="calculate()"><option value="3">Lv.3</option><option value="4">Lv.4</option><option value="5">Lv.5</option><option value="6">Lv.6</option></select></div><div><label>To</label><select class="p-to" onchange="calculate()"><option value="4">Lv.4</option><option value="5" selected>Lv.5</option><option value="6">Lv.6</option><option value="7">Lv.7</option></select></div><div><label>Qty</label><input type="number" class="p-qty" oninput="calculate()"></div><button onclick="this.parentElement.parentElement.remove(); calculate();">X</button></div>`;
    container.appendChild(div);
  }

  function calcSprint() {
    const bonus = parseFloat(document.getElementById('researchBonus').value) || 0;
    const target = document.getElementById('sp-target').value;
    const row = document.querySelector(`.row-unit-time[data-lvl="${target}"]`);
    const baseSec = (parseFloat(row.querySelector('.u-min').value) * 60) + parseFloat(row.querySelector('.u-sec').value);

    // Tempo com bónus: TempoBase / (1 + Bónus%)
    const adjSec = baseSec / (1 + (bonus / 100));

    const totalSec = ((parseFloat(document.getElementById('sp-d').value) || 0) * 86400) +
            ((parseFloat(document.getElementById('sp-h').value) || 0) * 3600) +
            ((parseFloat(document.getElementById('sp-m').value) || 0) * 60) +
            (parseFloat(document.getElementById('sp-s').value) || 0);

    const qty = totalSec > 0 && adjSec > 0 ? Math.floor(totalSec / adjSec) : 0;
    document.getElementById('sp-res-qty').innerText = qty.toLocaleString();
    document.getElementById('sp-res-pts').innerText = (qty * unitPoints[target]).toLocaleString();
  }

  function removeLineup(id) { document.getElementById('lineup-' + id).remove(); calculate(); }

  function calculate() {
    let globalTotal = 0;

    // Tribal Raid
    const stam = parseFloat(document.getElementById('staminaTotal').value) || 0;
    const cost = parseFloat(document.getElementById('staminaCost').value) || 5;
    const kills = Math.floor(stam / cost);
    const s1 = kills * (parseFloat(document.getElementById('tribeTier').value) || 0);
    document.getElementById('resKills').innerText = kills;
    document.getElementById('resPoints').innerText = s1.toLocaleString();
    document.getElementById('sub-st1').innerText = s1.toLocaleString();
    globalTotal += s1;

    // Day 3
    let gatherTotal = 0;
    document.querySelectorAll('[id^="lineup-"]').forEach(el => {
      const lvl = el.querySelector('.l-lvl').value;
      const isRich = el.querySelector('.l-rich').checked;
      const rounds = parseFloat(el.querySelector('.l-rounds').value) || 0;
      const speed = parseFloat(el.querySelector('.l-speed').value) || 0;
      const comp = parseFloat(el.querySelector('.l-comp').value) || 0;
      const baseMins = isRich ? richTimes[lvl] : baseTimes[lvl];
      const resourceAmount = isRich ? resRich[lvl] : resNormal[lvl];
      const ptsPerNode = resourceAmount / 100;
      const totalLineupPts = ptsPerNode * rounds;
      const finalMinsPerRound = baseMins / ((1 + speed/100) * (1 + comp/100));
      const totalMins = finalMinsPerRound * rounds;
      const h = Math.floor(totalMins / 60);
      const m = Math.round(totalMins % 60);
      el.querySelector('.l-time').innerText = (h >= 24) ? `${Math.floor(h/24)}d ${h%24}h ${m}m` : `${h}h ${m}m`;
      el.querySelector('.l-pts').innerText = Math.round(totalLineupPts).toLocaleString();
      gatherTotal += totalLineupPts;
    });
    document.getElementById('sub-st3').innerText = Math.round(gatherTotal).toLocaleString();
    globalTotal += gatherTotal;

    // Promotions
    let s5 = 0;
    document.querySelectorAll('[id^="promo-"]').forEach(el => {
      const diff = unitPoints[el.querySelector('.p-to').value] - unitPoints[el.querySelector('.p-from').value];
      s5 += (diff + 1) * (parseFloat(el.querySelector('.p-qty').value) || 0);
    });

    // Generic stages
    const stages = ['st2', 'st4', 'st5'];
    stages.forEach(sid => {
      let sub = (sid === 'st5' ? s5 : 0);
      document.getElementById(sid).querySelectorAll('.mge-val').forEach(i => sub += (parseFloat(i.dataset.pts) * (parseFloat(i.value) || 0)));
      if(sid === 'st4') {
        document.querySelectorAll('#st4 .row-speed').forEach(r => {
          const tMin = ((parseFloat(r.querySelector('.t-d').value) || 0) * 1440) + ((parseFloat(r.querySelector('.t-h').value) || 0) * 60) + (parseFloat(r.querySelector('.t-m').value) || 0);
          sub += tMin * 30;
        });
      }
      document.getElementById('sub-' + sid).innerText = Math.round(sub).toLocaleString();
      globalTotal += sub;
    });

    document.getElementById('totalPoints').innerText = Math.round(globalTotal).toLocaleString();
    calcSprint();
  }

  window.onload = () => { addLineup(); calculate(); };
</script>
